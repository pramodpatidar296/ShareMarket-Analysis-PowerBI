## Calendar Table
Calendar_Table = 
VAR AllDates =
    UNION (
        SELECTCOLUMNS (
            Booking_Sales,
            "Date", Booking_Sales[A/c. Date]
        ),
        SELECTCOLUMNS (
            Dispatch_Sales,
            "Date", Dispatch_Sales[A/c.Date]
        )
    )

VAR MinDate =
    MINX ( AllDates, [Date] )

VAR MaxDate =
    MAXX ( AllDates, [Date] )

RETURN
ADDCOLUMNS (
    CALENDAR ( MinDate, MaxDate ),

    /* ---------- Date Keys ---------- */
    "DateKey", FORMAT ( [Date], "yyyyMMdd" ),
    "Year", YEAR ( [Date] ),
    "MonthNumber", MONTH ( [Date] ),
    "MonthName", FORMAT ( [Date], "MMMM" ),
    "Day", DAY ( [Date] ),

    /* ---------- Week ---------- */
    "WeekdayNumber", WEEKDAY ( [Date], 2 ),          -- Monday = 1
    "WeekdayNameShort", FORMAT ( [Date], "ddd" ),
    "WeekStart", [Date] - ( WEEKDAY ( [Date], 2 ) - 1 ),
    "WeekEnd", [Date] - ( WEEKDAY ( [Date], 2 ) - 1 ) + 6,
    "YearWeek", YEAR ( [Date] ) * 100 + WEEKNUM ( [Date], 2 ),

    /* ---------- Calendar Quarter ---------- */
    "QuarterNumber", QUARTER ( [Date] ),
    "QuarterName", "Q" & FORMAT ( QUARTER ( [Date] ), "0" ),
    "YearQuarter", FORMAT ( [Date], "yyyy" ) & "Q" & QUARTER ( [Date] ),

    /* ========== Financial Year (INDIA) ========== */

    -- Financial Year Number
    "Financial_Year",
        IF (
            MONTH ( [Date] ) >= 4,
            YEAR ( [Date] ),
            YEAR ( [Date] ) - 1
        ),

    -- Financial Year Label (2024-25)
    "Financial_Year_Label",
        VAR FY =
            IF ( MONTH ( [Date] ) >= 4, YEAR ( [Date] ), YEAR ( [Date] ) - 1 )
        RETURN
            FORMAT ( FY, "0000" ) & "-" & FORMAT ( FY + 1, "00" ),

    -- Financial Month Number (April = 1 … March = 12)
    "FY_Month_Number",
        MOD ( MONTH ( [Date] ) + 8, 12 ) + 1,

    -- Financial Month Name
    "FY_Month_Name",
        FORMAT ( [Date], "MMM" ),

    -- Financial Quarter
    "FY_Quarter",
        "Q" &
        SWITCH (
            TRUE (),
            MONTH ( [Date] ) IN { 4, 5, 6 }, 1,
            MONTH ( [Date] ) IN { 7, 8, 9 }, 2,
            MONTH ( [Date] ) IN { 10, 11, 12 }, 3,
            4
        )
)

15.
Booking_Qty QTD = 
VAR TodayDate = TODAY()

-- Financial Year start year
VAR FYStartYear =
    IF ( MONTH ( TodayDate ) >= 4,
        YEAR ( TodayDate ),
        YEAR ( TodayDate ) - 1
    )

-- Financial month number (Apr = 1 … Mar = 12)
VAR FYMonth =
    MOD ( MONTH ( TodayDate ) + 8, 12 ) + 1

-- Financial Quarter number
VAR FYQuarter =
    INT ( ( FYMonth - 1 ) / 3 ) + 1

-- First FY month of the quarter
VAR FirstFYMonth =
    ( FYQuarter - 1 ) * 3 + 1

-- Convert FY month → calendar month
VAR FirstCalendarMonth =
    MOD ( FirstFYMonth + 2, 12 ) + 1

-- Calendar year of quarter start
VAR QuarterStartYear =
    IF ( FirstCalendarMonth >= 4, FYStartYear, FYStartYear + 1 )

VAR QuarterStartDate =
    DATE ( QuarterStartYear, FirstCalendarMonth, 1 )

RETURN
CALCULATE (
    [Total_Booking_Qty (Metric Ton)],
    FILTER (
        ALL ( 'Calendar_Table'[Date] ),
        'Calendar_Table'[Date] >= QuarterStartDate
            && 'Calendar_Table'[Date] <= TodayDate
    )
)

    IF ( FYQuarter > 1, FYQuarter - 1, 4 )

VAR PrevFY =
    IF ( FYQuarter > 1, FY, FY - 1 )

-- First month of previous FY quarter (FY month number)

    ( PrevFYQuarter - 1 ) * 3 + 1

-- Convert FY month → calendar month
VAR PrevCalendarMonth =
    MOD ( PrevFirstFYMonth + 2, 12 ) + 1

-- Calendar year of previous quarter start
VAR PrevQuarterStartYear =
    IF ( PrevCalendarMonth >= 4, PrevFY, PrevFY + 1 )

VAR PrevQuarterStart =
    DATE ( PrevQuarterStartYear, PrevCalendarMonth, 1 )

VAR PrevQuarterEnd =
    EOMONTH ( PrevQuarterStart, 2 )

VAR Vlu =
    CALCULATE (
        [Total_Booking_Qty (Metric Ton)],
        FILTER (
            ALL ( 'Calendar_Table'[Date] ),
            'Calendar_Table'[Date] >= PrevQuarterStart
                && 'Calendar_Table'[Date] <= PrevQuarterEnd
        )
    )

RETURN
IF (
    ISBLANK ( Vlu ) || Vlu = 0,
    "-",
    " " & FORMAT ( Vlu, "#,##,##,##,###" )
)

Booking_Qty_Card_Display_Quarter = 
VAR TodayDate = TODAY()

-- Financial Year
VAR FY =
    IF ( MONTH ( TodayDate ) >= 4,
        YEAR ( TodayDate ),
        YEAR ( TodayDate ) - 1
    )

-- Financial Month & Quarter
VAR FYMonth = MOD ( MONTH ( TodayDate ) + 8, 12 ) + 1
VAR FYQuarter = INT ( ( FYMonth - 1 ) / 3 ) + 1
VAR QuarterLabel = "Q" & FYQuarter

-- Current QTD
VAR CurrentQuarterSales = [Booking_Qty QTD]

-- Current Quarter Start
VAR CurrQuarterStartMonth =
    SWITCH(
        TRUE(),
        FYQuarter = 1, 4,
        FYQuarter = 2, 7,
        FYQuarter = 3, 10,
        1
    )

VAR CurrQuarterYear =
    IF ( CurrQuarterStartMonth = 1, FY + 1, FY )

VAR CurrQuarterStart =
    DATE ( CurrQuarterYear, CurrQuarterStartMonth, 1 )

-- Previous Quarter (Same FY)
VAR PrevQuarter =
    IF ( FYQuarter > 1, FYQuarter - 1, 4 )

VAR PrevQuarterFY =
    IF ( FYQuarter > 1, FY, FY - 1 )

VAR PrevQuarterStartMonth =
    SWITCH(
        TRUE(),
        PrevQuarter = 1, 4,
        PrevQuarter = 2, 7,
        PrevQuarter = 3, 10,
        1
    )

VAR PrevQuarterYear =
    IF ( PrevQuarterStartMonth = 1, PrevQuarterFY + 1, PrevQuarterFY )

VAR PrevQuarterStart =
    DATE ( PrevQuarterYear, PrevQuarterStartMonth, 1 )

-- SAME PERIOD
VAR PrevQuarterSameDate =
    DATE(
        YEAR(PrevQuarterStart),
        MONTH(PrevQuarterStart) + (MONTH(TodayDate) - MONTH(CurrQuarterStart)),
        DAY(TodayDate)
    )

VAR PrevQuarterEnd =
    MIN(PrevQuarterSameDate, EOMONTH(PrevQuarterStart, 2))

VAR PrevQuarterSales =
    CALCULATE (
        [Total_Booking_Qty (Metric Ton)],
        FILTER (
            ALL ( 'Calendar_Table'[Date] ),
            'Calendar_Table'[Date] >= PrevQuarterStart
                && 'Calendar_Table'[Date] <= PrevQuarterEnd
        )
    )

-- % Diff
VAR PctDiff =
    DIVIDE ( CurrentQuarterSales - PrevQuarterSales, PrevQuarterSales )

VAR Arrow =
    IF ( PctDiff >= 0, "🟢 ↑", "🔴 ↓" )

RETURN
"FY(" & QuarterLabel & ") " &
FORMAT ( CurrentQuarterSales, "#,##,##,##,###" ) &
" vs FY(Q" & PrevQuarter & ") " &
FORMAT ( PrevQuarterSales, "#,##,##,##,###" ) &
UNICHAR(10) &
FORMAT ( PctDiff, "0.0%" ) & " " & Arrow

Current Week Booking Qty = 
VAR TodayDate =
    CALCULATE (
        MAX ( 'Calendar_Table'[Date] ),
        ALL ( 'Calendar_Table' ),
        'Calendar_Table'[Date] <= TODAY()
    )

VAR WeekStart =
    TodayDate - ( WEEKDAY ( TodayDate, 2 ) - 1 )

VAR WeekDayNum =
    SELECTEDVALUE ( 'Calendar_Table'[WeekdayNumber] )

VAR ColumnDate =
    WeekStart + WeekDayNum - 1

RETURN
IF (
    -- 🔹 DAY COLUMNS (Mon–Sun)
    ISINSCOPE ( 'Calendar_Table'[WeekdayNameShort] ),

    IF (
        ColumnDate > TodayDate,
        BLANK(),
        CALCULATE (
            SUM ( 'Booking_Sales'[BookingQuantity] ) / 1000,
            ALL ( 'Calendar_Table'[Date] ),
            'Calendar_Table'[Date] = ColumnDate
        )
    ),

    -- 🔹 ROW TOTAL (Mon → Today)
    CALCULATE (
        SUM ( 'Booking_Sales'[BookingQuantity] ) / 1000,
        FILTER (
            ALL ( 'Calendar_Table'[Date] ),
            'Calendar_Table'[Date] >= WeekStart
                && 'Calendar_Table'[Date] <= TodayDate
        )
    )
)
